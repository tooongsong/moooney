# Best Card Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Record each credit card's cashback rates and show one page that names the best card per category.

**Architecture:** Rates live in one nullable `rewards` jsonb column on `payment_methods`. A pure module, `src/lib/bestCard.ts`, validates rates, derives the per-category winners, and rewrites keys when a category is renamed or deleted. Two screens read it: an editor on the credit card's detail page and a read-only `/accounts/best-card` page.

**Tech Stack:** Next.js 16 App Router (server components + server actions), Drizzle ORM on Supabase Postgres, Tailwind v4, `node --test` for lib tests.

**Spec:** `docs/superpowers/specs/2026-11-19-best-card-design.md`

## Global Constraints

- Rates are percentages as plain numbers (`3` means 3%), range 0–100, at most two decimal places.
- `rewards` is `null` when a card has no rules (base 0 and no category rates).
- Only `type = 'credit_card'` accounts have rewards; archived cards are ignored.
- Category keys are the category display name, the same string `transactions.category` holds.
- No new dependencies.
- Queries against `payment_methods` that only need some fields must pass `columns:` (Supabase egress limit).
- Lib tests import with the `.ts` extension and the lib file must contain only type-strippable TypeScript (no enums, no parameter properties) — `npm test` runs `node --test src/lib/*.test.ts` with no build step.
- Design direction: the rate is the graphic, labels are small, radius 16 / 24 / pill only, no shadows, accent on the single highest rate only.
- **Deploy order:** the migration in Task 2 must be applied to the production database before any commit from Task 2 onward is pushed. Drizzle selects every declared column, so code that declares `rewards` fails against a table that lacks it. Applying the migration and pushing both need the owner's explicit go-ahead.

```ts
// Credit-card reward rates and the "which card for this category" answer.
// Rates are percentages as plain numbers: 3 means 3%.

export interface CardRewards {
  base: number;
  categories: Record<string, number>;
}

export interface RewardCard {
  id: string;
  name: string;
  rewards: CardRewards | null;
}

export interface BestCardPick {
  rate: number;
  cards: { id: string; name: string }[];
}

export interface BestCardRow extends BestCardPick {
  category: string;
}

export interface BestCards {
  rows: BestCardRow[];
  fallback: BestCardPick | null;
}

/** 0–100 with at most two decimals. The epsilon absorbs float noise (1.15 * 100). */
function isRate(n: unknown): n is number {
  return typeof n === 'number' && Number.isFinite(n) && n >= 0 && n <= 100
    && Math.abs(Math.round(n * 100) - n * 100) < 1e-6;
}

const RATE_ERROR = 'A rate must be between 0 and 100, with at most two decimals.';

/** Validates what the editor sends. `rewards: null` means "store nothing". */
export function cleanRewards(
  input: { base: number; categories: Record<string, number> },
): { ok: true; rewards: CardRewards | null } | { ok: false; error: string } {
  if (!isRate(input.base)) return { ok: false, error: RATE_ERROR };
  const categories: Record<string, number> = {};
  for (const [name, rate] of Object.entries(input.categories)) {
    const trimmed = name.trim();
    if (!trimmed) return { ok: false, error: 'A category rate needs a category.' };
    if (!isRate(rate)) return { ok: false, error: RATE_ERROR };
    categories[trimmed] = rate;
  }
  if (input.base === 0 && Object.keys(categories).length === 0) return { ok: true, rewards: null };
  return { ok: true, rewards: { base: input.base, categories } };
}

/**
 * One row per category in `categories` that at least one card has a rate for,
 * plus the best base rate as the "everything else" fallback. A card without a
 * rate for a category competes with its base rate.
 */
export function bestCards(cards: RewardCard[], categories: readonly string[]): BestCards {
  const ruled = cards.filter((c): c is RewardCard & { rewards: CardRewards } => c.rewards !== null);
  if (ruled.length === 0) return { rows: [], fallback: null };

  const pick = (rateOf: (r: CardRewards) => number): BestCardPick => {
    const rate = Math.max(...ruled.map((c) => rateOf(c.rewards)));
    return {
      rate,
      cards: ruled.filter((c) => rateOf(c.rewards) === rate).map(({ id, name }) => ({ id, name })),
    };
  };

  const rows = categories
    .filter((cat) => ruled.some((c) => c.rewards.categories[cat] !== undefined))
    .map((cat) => ({ category: cat, ...pick((r) => r.categories[cat] ?? r.base) }))
    .sort((a, b) => b.rate - a.rate || a.category.localeCompare(b.category));

  return { rows, fallback: pick((r) => r.base) };
}

/** Same object back when `from` is not a key, so callers can skip the write. */
export function renameRewardCategory(rewards: CardRewards, from: string, to: string): CardRewards {
  const rate = rewards.categories[from];
  if (rate === undefined) return rewards;
  const categories = { ...rewards.categories };
  delete categories[from];
  // A rate already stored under the new name wins over the renamed one.
  if (categories[to] === undefined) categories[to] = rate;
  return { base: rewards.base, categories };
}

/** Same object back when `name` is not a key; null when nothing is left to store. */
export function removeRewardCategory(rewards: CardRewards, name: string): CardRewards | null {
  if (rewards.categories[name] === undefined) return rewards;
  const categories = { ...rewards.categories };
  delete categories[name];
  if (rewards.base === 0 && Object.keys(categories).length === 0) return null;
  return { base: rewards.base, categories };
}

export function formatRate(rate: number): string {
  return `${rate}%`;
}
```

```ts
export async function updateCardRewards(
  id: string,
  input: { base: number; categories: Record<string, number> },
): Promise<{ success: true } | { success: false; error: string }> {
  const user = await getUser();
  const cleaned = cleanRewards(input);
  if (!cleaned.ok) return { success: false, error: cleaned.error };

  const mine = and(eq(paymentMethods.id, id), eq(paymentMethods.userId, user.id));
  const account = await db.query.paymentMethods.findFirst({ where: mine, columns: { type: true } });
  if (!account || account.type !== 'credit_card') {
    return { success: false, error: 'Rewards can only be set on a credit card.' };
  }

  await db.update(paymentMethods).set({ rewards: cleaned.rewards, updatedAt: new Date() }).where(mine);
  revalidatePath('/accounts');
  revalidatePath(`/accounts/${id}`);
  revalidatePath('/accounts/best-card');
  return { success: true };
}
```

```ts
/** Applies `change` to every card's reward rules, writing only the cards it altered. */
async function rewriteCardRewards(userId: string, change: (r: CardRewards) => CardRewards | null) {
  const cards = await db.query.paymentMethods.findMany({
    where: eq(paymentMethods.userId, userId),
    columns: { id: true, rewards: true },
  });
  for (const card of cards) {
    if (!card.rewards) continue;
    const next = change(card.rewards);
    if (next !== card.rewards) {
      await db.update(paymentMethods).set({ rewards: next }).where(eq(paymentMethods.id, card.id));
    }
  }
  revalidatePath('/accounts/best-card');
}
```

```tsx
'use client';

import { useState } from 'react';
import { Loader2, Plus, X } from 'lucide-react';
import { toast } from 'sonner';
import { Input } from '@/components/ui/input';
import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from '@/components/ui/select';
import { updateCardRewards } from '@/app/actions/accounts';
import type { CardRewards } from '@/lib/bestCard';

interface Row { category: string; rate: string }

const onlyRate = (v: string) => v.replace(/[^0-9.]/g, '');
const FIELD = 'rounded-lg border-line bg-paper-card text-sm h-9 tabular-nums';
const LABEL = 'text-[10px] font-semibold uppercase tracking-widest text-ink-soft';

export function CardRewardsEditor({
  accountId, initial, categories,
}: { accountId: string; initial: CardRewards | null; categories: string[] }) {
  const [base, setBase] = useState(initial ? String(initial.base) : '');
  const [rows, setRows] = useState<Row[]>(
    () => Object.entries(initial?.categories ?? {}).map(([category, rate]) => ({ category, rate: String(rate) })),
  );
  const [newCategory, setNewCategory] = useState('');
  const [newRate, setNewRate] = useState('');
  const [saving, setSaving] = useState(false);

  const unused = categories.filter((c) => !rows.some((r) => r.category === c));

  function addRow() {
    if (!newCategory || !newRate) return;
    setRows((prev) => [...prev, { category: newCategory, rate: newRate }]);
    setNewCategory('');
    setNewRate('');
  }

  async function save() {
    setSaving(true);
    const result = await updateCardRewards(accountId, {
      base: base === '' ? 0 : Number(base),
      categories: Object.fromEntries(rows.map((r) => [r.category, Number(r.rate)])),
    });
    setSaving(false);
    if (result.success) toast.success('Rewards saved');
    else toast.error(result.error);
  }

  return (
    <div className="space-y-3">
      <div className="flex items-center justify-between gap-3">
        <label htmlFor="rewards-base" className={LABEL}>Base · everything else</label>
        <span className="flex items-center gap-1.5">
          <Input id="rewards-base" inputMode="decimal" value={base} placeholder="0"
            onChange={(e) => setBase(onlyRate(e.target.value))} className={`${FIELD} w-20 text-right`} />
          <span className="text-sm text-ink-soft">%</span>
        </span>
      </div>

      {rows.map((row, i) => (
        <div key={row.category} className="flex items-center justify-between gap-3">
          <span className="min-w-0 truncate text-sm font-semibold text-ink">
            {row.category}
            {!categories.includes(row.category) && (
              <span className="ml-2 text-[10px] font-semibold uppercase tracking-widest text-ink-soft">Unused</span>
            )}
          </span>
          <span className="flex items-center gap-1.5 shrink-0">
            <Input inputMode="decimal" value={row.rate} aria-label={`${row.category} rate`}
              onChange={(e) => setRows((prev) => prev.map((r, j) => (j === i ? { ...r, rate: onlyRate(e.target.value) } : r)))}
              className={`${FIELD} w-20 text-right`} />
            <span className="text-sm text-ink-soft">%</span>
            <button type="button" aria-label={`Remove ${row.category}`}
              onClick={() => setRows((prev) => prev.filter((_, j) => j !== i))}
              className="w-9 h-9 rounded-full flex items-center justify-center text-ink-soft hover:bg-sand transition-colors">
              <X className="h-4 w-4" />
            </button>
          </span>
        </div>
      ))}

      {unused.length > 0 && (
        <div className="flex items-center gap-2 pt-1">
          <Select value={newCategory} onValueChange={setNewCategory}>
            <SelectTrigger className="rounded-lg border-line bg-paper-card h-9 text-sm flex-1 min-w-0" aria-label="Category">
              <SelectValue placeholder="Add category" />
            </SelectTrigger>
            <SelectContent>
              {unused.map((c) => <SelectItem key={c} value={c}>{c}</SelectItem>)}
            </SelectContent>
          </Select>
          <Input inputMode="decimal" value={newRate} placeholder="0" aria-label="New category rate"
            onChange={(e) => setNewRate(onlyRate(e.target.value))} className={`${FIELD} w-20 text-right`} />
          <span className="text-sm text-ink-soft">%</span>
          <button type="button" aria-label="Add category rate" onClick={addRow} disabled={!newCategory || !newRate}
            className="w-9 h-9 shrink-0 rounded-full bg-ink text-paper flex items-center justify-center disabled:opacity-30">
            <Plus className="h-4 w-4" />
          </button>
        </div>
      )}

      <button type="button" onClick={save} disabled={saving}
        className="w-full h-12 rounded-full bg-accent text-white text-sm font-semibold uppercase tracking-widest flex items-center justify-center disabled:opacity-50">
        {saving ? <Loader2 className="h-5 w-5 animate-spin" /> : 'Save rewards'}
      </button>
    </div>
  );
}
```

- [ ] **Step 1: Write the page**

Create `src/app/accounts/best-card/page.tsx`:

```tsx
import Link from 'next/link';
import { redirect } from 'next/navigation';
import { ChevronLeft } from 'lucide-react';
import { BottomNav } from '@/components/BottomNav';
import { HeaderIconButton } from '@/components/HeaderIconButton';
import { PageHeader } from '@/components/PageHeader';
import { getAccountBalances } from '@/app/actions/accounts';
import { getAllCategories } from '@/app/actions/manage';
import { bestCards, formatRate, type BestCardPick } from '@/lib/bestCard';

function Row({ label, pick, accent }: { label: string; pick: BestCardPick; accent: boolean }) {
  return (
    <div className="flex items-end justify-between gap-4 py-5 border-b border-line">
      <div className="min-w-0 pb-1">
        <p className="text-[10px] font-semibold uppercase tracking-widest text-ink-soft mb-1">{label}</p>
        <p className="text-base font-semibold text-ink truncate">{pick.cards.map((c) => c.name).join(' · ')}</p>
      </div>
      <p className={`shrink-0 text-5xl font-bold tracking-tighter leading-none tabular-nums ${accent ? 'text-accent' : 'text-ink'}`}>
        {formatRate(pick.rate)}
      </p>
    </div>
  );
}

export default async function BestCardPage() {
  const [accounts, categories] = await Promise.all([getAccountBalances(), getAllCategories()]);
  const cards = accounts.filter((a) => a.type === 'credit_card');
  if (cards.length === 0) redirect('/accounts');

  const { rows, fallback } = bestCards(cards, categories);

  return (
    <div className="d-max-md max-lg:max-w-md mx-auto px-6 min-h-screen bg-paper pb-28">
      <PageHeader
        left={<HeaderIconButton href="/accounts" className="-ml-2" aria-label="Back to accounts"><ChevronLeft /></HeaderIconButton>}
        title="Best card"
      />

      {fallback === null ? (
        <div className="pt-10 pb-16">
          <p className="text-3xl font-bold tracking-tighter text-ink-faint leading-tight">
            NO REWARDS<br />YET.
          </p>
          <Link href={`/accounts/${cards[0].id}`}
            className="inline-block mt-6 text-xs font-bold uppercase tracking-widest text-ink underline underline-offset-4">
            Add rates to {cards[0].name}
          </Link>
        </div>
      ) : (
        <div>
          {rows.map((row, i) => (
            <Row key={row.category} label={row.category} pick={row} accent={i === 0} />
          ))}
          <Row label="Everything else" pick={fallback} accent={rows.length === 0} />
        </div>
      )}

      <BottomNav />
    </div>
  );
}
```

Note: `HeaderIconButton` spreads extra props onto a `<button>` only; when `href` is set it renders a `Link` and drops them. If `aria-label` does not reach the DOM, leave it — the header title names the page — and do not change `HeaderIconButton` in this task.
